70 Parameters getParameters() const {
71 // Значения по умолчанию.
72 double duration = 1.;
73 double spatial_step = 0.05;
74 bool service_mode = false;
75 std::string file_name = "data.txt";
76 // Разбор аргументов при помощи getopt.
77 int option = 0;
78 while ((option = getopt(_argument_count, _argument_strings, "t:x:o:s")) != -1) {
79 switch (option) {
80 case 't':
81 duration = atof(optarg);
82 if (duration <= 0.) {
83 throw std::invalid_argument("Parameters: duration must be positive");
84 }
85 break;
86 case 'x':
87 spatial_step = atof(optarg);
88 if (spatial_step <= 0.) {
89 throw std::invalid_argument("Parameters: spatial step must be positive");
90 }
91 break;
92 case 'o':
93 file_name = optarg;
94 break;
95 case 's':
96 service_mode = true;
97 break;
98 default:
99 break;
100 }
101 }
102 // Создание объекта параметров.
103 Parameters parameters;
104 // Заполнение полей структуры значениями параметров.
105 parameters.output_file = file_name;
106 parameters.duration = duration;
107 parameters.spatial_step = spatial_step;
108 parameters.no_output = service_mode;
109 // Начальные и граничные условия.
110 parameters.conditions.force = zeroForce; // harmonicForce
111 parameters.conditions.begin = nonZeroBeginCondition; // zeroBeginCondition
112 parameters.conditions.left = Boundary{ BoundaryType::first, 0. };
113 parameters.conditions.right = Boundary{ BoundaryType::first, 0. };
114 return parameters;
115 }
116
117 private:
118
119 const int _argument_count;
120 char** _argument_strings;
