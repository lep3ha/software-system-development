47 inline double calculateNode(double f, unsigned n) const noexcept {
48 const double inner_sum = _current.get()[n + 1] - 2. * _current.get()[n] + _current.get()[n - 1];
49 const double time_multiplier = pow(_phase_velocity, 2.) * inner_sum / pow(_spatial_step, 2.);
50 const double last_term = (time_multiplier + f) * pow(_time_step, 2.);
51 const double result = 2. * _current.get()[n] - _previous.get()[n] + last_term;
52 return result;
53 }

60 void calculateNodes(unsigned long b, unsigned long s, const Parameters& p, Plotter& o) noexcept {
61 // Цикл по времени моделирования.
62 unsigned long current_time_step = 1u;
63 while (current_time_step <= _time_steps_number) {
64 // Цикл по узлам сетки.
65 for (unsigned long i = b; i < b + s; i++) {
66 if (i == 0u) {
67 _next.get()[i] = calculateLeftBoundary(p.conditions.left);
68 } else if (i == _size - 1) {
69 _next.get()[i] = calculateRightBoundary(p.conditions.right);
70 } else {
71 // Расчёт значений по явной вычислительной схеме.
72 const double force = p.conditions.force(i * p.spatial_step, _time_step * current_time_step);
73 _next.get()[i] = calculateNode(force, i);
74 }
75 }
76 // Вывод решения.
77 unsigned plot_step = 0.05 / _time_step; // для "реального" масштаба времени анимации
78 if (!p.no_output && current_time_step % plot_step == 0) {
79 o.plot(_next.get(), _size);
80 }
81
82 // Обмен указателями для повышения быстродействия.
83 _current.swap(_next); // в last теперь текущий слой, в current теперь слой j
84 _previous.swap(_next); // в previous теперь слой j, в current теперь слой j-1
85
86 // Следующий шаг по времени.
87 current_time_step++;
88 }
89 }

122 _size = round(string_length / p.spatial_step) + 1ul; // число узлов сетки
123 _next = std::shared_ptr<double[]>(new double[_size]);
124 for (unsigned long i = 0lu; i < _size; i++) {
125 const double x = i * p.spatial_step;
126 _next.get()[i] = p.conditions.begin(x); // начальные условия
127 }
128 _current = std::shared_ptr<double[]>(new double[_size]);
129 _previous = std::shared_ptr<double[]>(new double[_size]);
130 for (unsigned long i = 0lu; i < _size; i++) {
131 _current.get()[i] = _next.get()[i];
132 _previous.get()[i] = _next.get()[i];
133 }
